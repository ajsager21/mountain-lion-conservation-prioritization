# --- Environment note ---
# This project was built after resolving a persistent kernel-crash conflict between
# pip-installed and conda-installed GDAL builds (geopandas/rasterio/rasterstats).
# If you hit silent kernel crashes on zonal/raster operations, create a clean
# conda-forge environment rather than pip-installing into an existing Anaconda base:
#
#   conda create -n geo python=3.11 geopandas rasterio pygbif scikit-learn xgboost \
#       scipy pulp networkx matplotlib jupyter -c conda-forge -y
#   conda activate geo
#   python -m ipykernel install --user --name geo --display-name "Python (geo)"
#
# rasterstats was deliberately abandoned in this project due to this exact conflict.
# Raster value extraction is done with plain rasterio point sampling + scipy
# nearest-neighbor fill instead — do not reintroduce rasterstats.
